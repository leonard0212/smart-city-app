CREATE view wDetectionsInfrastructure  as  select d.Id as DetectionId,d.CreatedAt,   EOMONTH(d.CreatedAt) as CreatedMonth,   cast(1 as int) as DetectionsCount,   d.Category,d.Description,d.NeedIntervention,d.Severity,d.lat,d.Lng,   d.ResolutionStatus,d.ReportingDate,r.Id as RawDataId,   JSON_VALUE(Body, '$.ImageUrl') AS ImageUrl,      JSON_VALUE(Body, '$.ImageUrlPreview') AS ImageUrlPreview,   ext.Type,   ext.Dimension,   ext.Diameter,   ext.Depth,   r.ProcessedsFileId,   r.PreviewFileId,      (select Path from AppFile where Id=r.PreviewFileId) as PreviewFilePath,   (select Path from AppFile where Id=r.ProcessedsFileId) as ProcessedsFilePath,     (select [Url] from AzureUrl where Id=r.PreviewFileId) as PreviewFileUrl,   (select [Url] from AzureUrl where Id=r.ProcessedsFileId) as ProcessedsFileUrl      from detection d inner join RawDetection r    on r.Id=d.RawDataId    inner join [SmartCity.Logging].dbo.MessageBrokerDataLogs l on l.MessageId=r.ServiceBusMessageId    inner join RoadInfrastructureDetection ext on ext.Id=d.Id
  go
create view wDetectionsTrash  as  select d.Id as DetectionId,d.CreatedAt,   EOMONTH(d.CreatedAt) as CreatedMonth,   cast(1 as int) as DetectionsCount,   d.Category,d.Description,d.NeedIntervention,d.Severity,d.lat,d.Lng,   d.ResolutionStatus,d.ReportingDate,r.Id as RawDataId,   JSON_VALUE(Body, '$.ImageUrl') AS ImageUrl,      JSON_VALUE(Body, '$.ImageUrlPreview') AS ImageUrlPreview,   ext.AssetStatus,   ext.TrashAssetCategory,   ext.TrashDescription,   ext.TrashQuantity,   ext.TrashCondition,   ext.TrashVolumType,   r.ProcessedsFileId,   r.PreviewFileId,      (select Path from AppFile where Id=r.PreviewFileId) as PreviewFilePath,   (select Path from AppFile where Id=r.ProcessedsFileId) as ProcessedsFilePath,     (select [Url] from AzureUrl where Id=r.PreviewFileId) as PreviewFileUrl,   (select [Url] from AzureUrl where Id=r.ProcessedsFileId) as ProcessedsFileUrl      from detection d inner join RawDetection r    on r.Id=d.RawDataId    inner join [SmartCity.Logging].dbo.MessageBrokerDataLogs l on l.MessageId=r.ServiceBusMessageId    inner join TrashAssetsDetection ext on ext.Id=d.Id    
go
